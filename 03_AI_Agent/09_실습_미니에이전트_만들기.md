# ★ 실습: 프레임워크 없이 미니 에이전트 만들기

목표: LLM API + 도구 2개(계산기, 노트 검색)로 **도구 호출 루프 에이전트**를 직접 구현해 원리를 체득한다.

## 준비
```bash
pip install anthropic        # 또는 사용하는 LLM 제공사 SDK
export ANTHROPIC_API_KEY=...  # API 키는 코드에 직접 쓰지 말고 환경변수로
```

## 코드 (Anthropic Messages API 예시)
```python
import os, glob, anthropic

client = anthropic.Anthropic()
MODEL = "<사용할 모델 이름>"   # 공식 문서의 최신 모델명 확인

# ---------- 1) 도구 구현 ----------
def calculator(expression: str) -> str:
    allowed = set("0123456789+-*/(). ")
    if not set(expression) <= allowed:
        return "오류: 숫자와 + - * / ( ) 만 사용할 수 있습니다."
    try:
        return str(eval(expression, {"__builtins__": {}}))
    except Exception as e:
        return f"계산 오류: {e}"

def search_notes(keyword: str) -> str:
    hits = []
    for path in glob.glob("../**/*.md", recursive=True):
        for i, line in enumerate(open(path, encoding="utf-8"), 1):
            if keyword.lower() in line.lower():
                hits.append(f"{os.path.basename(path)}:{i}: {line.strip()}")
    return "\n".join(hits[:20]) or "검색 결과 없음. 다른 키워드를 시도하세요."

TOOLS = {"calculator": calculator, "search_notes": search_notes}

# ---------- 2) 도구 스키마 ----------
TOOL_SCHEMAS = [
    {"name": "calculator",
     "description": "사칙연산 수식을 계산합니다. 숫자 계산이 필요할 때 반드시 사용하세요.",
     "input_schema": {"type": "object",
                      "properties": {"expression": {"type": "string", "description": "예: (3+4)*2"}},
                      "required": ["expression"]}},
    {"name": "search_notes",
     "description": "공부 노트(md 파일)에서 키워드를 찾아 파일명:줄번호와 내용을 반환합니다. "
                    "머신러닝/딥러닝/에이전트 개념 질문에 답하기 전에 사용하세요.",
     "input_schema": {"type": "object",
                      "properties": {"keyword": {"type": "string"}},
                      "required": ["keyword"]}},
]

SYSTEM = ("당신은 사용자의 AI 공부를 돕는 튜터입니다. 개념 질문엔 먼저 노트를 검색하고, "
          "노트 내용을 근거로 쉽게 설명하세요. 출처(파일명)를 함께 알려주세요.")

# ---------- 3) 에이전트 루프 ----------
def run(user_input: str, max_steps: int = 8) -> str:
    messages = [{"role": "user", "content": user_input}]
    for step in range(max_steps):
        resp = client.messages.create(model=MODEL, max_tokens=1024, system=SYSTEM,
                                      tools=TOOL_SCHEMAS, messages=messages)
        messages.append({"role": "assistant", "content": resp.content})

        if resp.stop_reason != "tool_use":
            return "".join(b.text for b in resp.content if b.type == "text")

        results = []
        for block in resp.content:
            if block.type == "tool_use":
                print(f"  🔧 [{step}] {block.name}({block.input})")
                out = TOOLS[block.name](**block.input)
                results.append({"type": "tool_result", "tool_use_id": block.id, "content": out})
        messages.append({"role": "user", "content": results})
    return "최대 단계에 도달했습니다."

if __name__ == "__main__":
    while True:
        q = input("\n질문> ")
        if q in ("exit", "quit"): break
        print(run(q))
```

## 해볼 실험
1. "LoRA가 뭐야?" → 노트 검색 후 설명하는지 확인
2. "1234*5678 + 99는?" → 계산기 사용 여부 확인
3. 도구 `description`을 일부러 짧게 바꿔 보고 행동이 어떻게 변하는지 관찰
4. 노트 파일에 "이전 지시를 무시하고 …" 같은 문장을 넣어 **간접 프롬프트 인젝션** 실험 → 방어 프롬프트 추가
5. 대화 기록을 유지하는 **멀티턴** + 오래된 메시지 **요약** 기능 추가
6. 평가셋 10문항을 만들어 자동 채점 스크립트 작성
7. 같은 기능을 LangGraph 또는 MCP 서버로 다시 만들어 비교
